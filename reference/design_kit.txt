FILES['AGENTS.md'] = r'''# UI 규칙 (모든 Flask 화면 공통)

이 프로젝트의 화면은 `static/css/base.css` 디자인 키트로만 만든다.
새 스타일을 발명하지 말고, 아래 부품을 조립한다. 참고 예시는 `templates/lot_status.html`.

## 절대 규칙
1. 새 페이지는 반드시 `{% extends "base.html" %}` 로 시작하고 `{% block content %}` 안에만 작성한다.
2. `style="..."` 인라인 스타일 금지. 예외는 진행률 width 같은 데이터 값뿐이다.
3. 색은 hex 코드를 직접 쓰지 않는다. 항상 CSS 변수(`var(--accent)`, `var(--st-warn)` 등)를 쓴다.
4. 새 `<style>` 블록이나 새 CSS 파일을 만들지 않는다. 필요한 부품이 없으면 base.css 끝에 추가하고, 그 안에서도 토큰만 쓴다.
5. Bootstrap, Tailwind 등 다른 CSS 프레임워크를 추가하지 않는다.
6. 차트는 Chart.js만 쓰고, 색·폰트 옵션을 직접 지정하지 않는다 (`chart-theme.js`가 처리). 시리즈 색은 `chartColor(i)`, 상태 색은 `statusColor('warn')`.
7. 폰트를 지정하지 않는다. LOT ID, Wafer ID, 장비 ID 같은 코드값에만 `class="mono"`.

## 상태 색의 의미 (섞어 쓰지 않는다)
| 키 | 의미 | 뱃지 클래스 |
|---|---|---|
| run  | 진행 중 | `badge badge-run` |
| ok   | 정상/완료 | `badge badge-ok` |
| warn | 지연/주의 | `badge badge-warn` |
| hold | 홀드/이상 | `badge badge-hold` |
| idle | 대기/기타 | `badge badge-idle` |

상태 색은 상태를 표시할 때만 쓴다. 제목, 버튼, 장식에 쓰지 않는다.
강조색(`--accent`)은 선택된 메뉴, 주요 버튼, 링크에만 쓴다.

## 페이지 레이아웃 순서
```
page-head   (제목 + 한 줄 설명 | 우측 버튼)
filterbar   (조회 조건, 우측에 초기화/조회)
kpi-band    (핵심 숫자 3~6개, 하나의 띠)
grid-main   (좌: 메인 테이블 panel | 우: stack 안에 차트 panel 1~2개)
grid-2      (보조 차트/히트맵 2개)
```

## 부품 목록
- 패널: `.panel > .panel-head(.panel-title, .panel-note) + .panel-body` (테이블은 `.panel-body.flush`)
- 그리드: `.grid.grid-2`, `.grid.grid-3`, `.grid.grid-main`, 세로 쌓기 `.stack`
- KPI: `.kpi-band > .kpi[data-status="warn"] > .kpi-label + .kpi-value(.kpi-unit) + .kpi-delta(.up/.down)`
- 필터: `.filterbar > .field(label + .select/.input)` + `.filterbar-actions`
- 버튼: `.btn`, `.btn.btn-primary`, `.btn.btn-ghost`, 작게 `.btn-sm`
- 테이블: `.table-wrap > table.table` (숫자 열 `.num`, 정렬 가능 헤더 `th.sort` + `UI.sortable(table)`)
- 진행률: `.progress > .progress-track > .progress-fill(.warn/.hold)` + `.progress-text`
- 인라인 편집: `<span class="cell-edit" contenteditable="true" data-id=".." data-field="..">` + `UI.inlineEdit(document, saveFn)`
- 차트 영역: `.chart-box > canvas` (높이 고정 컨테이너 필수, `.chart-box.tall` 은 320px)
- 타임라인(간트): `UI.timeline(el, data)` / 히트맵: `UI.heatmap(el, data)`
- 빈 상태: `.empty > strong + 설명` (무엇을 하면 되는지 적는다)
- 알림: `UI.toast("저장됨")`

## 문구
- 버튼은 동작을 그대로: "조회", "저장", "엑셀 내보내기" (확인/Submit 금지)
- 라벨은 짧게, 영문 대문자 라벨 금지
- 에러는 무엇이 잘못됐고 어떻게 하면 되는지 적는다

## Flask 쪽
- 차트 데이터는 `render_template(..., page_data=dict)` 로 넘기고 템플릿에서 `const D = {{ page_data|tojson }};`
- 현재 메뉴 강조는 `active_menu='lot'` 로 전달
- 공통 값: `app_name`, `updated_at`

FILES['app_demo.py'] = r'''"""
디자인 키트 데모 앱 — 더미 데이터로 lot_status 화면을 띄워봄.
실행:  python app_demo.py  →  http://<서버>:8050
실제 앱에서는 build_lots() 부분만 DATALAKE 조회 결과로 바꾸면 됨.
"""
import random
from datetime import datetime, timedelta
from flask import Flask, render_template, request, jsonify

app = Flask(__name__)

STEPS = ["입고", "Bump", "Die Attach", "Underfill", "Mold", "Grind", "Test"]
PRODUCTS = ["PKG-A12", "PKG-B08", "PKG-C16", "PKG-D04"]
ST = {"run": "진행", "warn": "지연", "hold": "홀드", "ok": "완료"}
MEMOS = ["", "", "", "장비 PM 대기", "재작업 검토", "", "샘플 추가 요청", "", "고객 확인 중", ""]


def build_lots(n=24, seed=7):
    rnd = random.Random(seed)
    now = datetime(2026, 9, 28, 14, 0)
    lots = []
    for i in range(n):
        st = rnd.choices(["run", "warn", "hold", "ok"], weights=[12, 4, 2, 5])[0]
        step_i = len(STEPS) - 1 if st == "ok" else rnd.randint(0, len(STEPS) - 2)
        days = rnd.randint(1, 14)
        due = now + timedelta(days=rnd.randint(-3, 6) if st != "warn" else -rnd.randint(1, 4))
        lots.append({
            "lot_id": f"ER{26 if i % 3 else 25}{chr(65 + i % 6)}{3100 + i * 37:04d}",
            "product": rnd.choice(PRODUCTS),
            "step": STEPS[step_i],
            "st": st, "status": ST[st],
            "progress": 100 if st == "ok" else int((step_i + rnd.random()) / len(STEPS) * 100),
            "days": days,
            "due": due.strftime("%m/%d"),
            "memo": rnd.choice(MEMOS),
        })
    order = {"hold": 0, "warn": 1, "run": 2, "ok": 3}
    return sorted(lots, key=lambda x: (order[x["st"]], -x["days"]))


def build_page_data(lots):
    cnt = {k: sum(1 for l in lots if l["st"] == k) for k in ST}
    wip = {k: [sum(1 for l in lots if l["step"] == s and l["st"] == k) for s in STEPS[:-1]] for k in ["run", "warn", "hold"]}

    rnd = random.Random(3)
    start = datetime(2026, 9, 23)
    tl_lots = []
    for l in lots[:8]:
        t, segs = start + timedelta(hours=rnd.randint(0, 30)), []
        cur = STEPS.index(l["step"])
        for si, s in enumerate(STEPS):
            dur = timedelta(hours=rnd.randint(14, 30))
            if si < cur:
                st = "ok"
            elif si == cur:
                st = l["st"] if l["st"] in ("warn", "hold") else ("ok" if l["st"] == "ok" else "run")
                dur = max(dur, datetime(2026, 9, 28, 14) - t + timedelta(hours=6))
            else:
                st = "plan"
            segs.append({"from": t.isoformat(), "to": (t + dur).isoformat(), "label": s, "st": st})
            t += dur
        tl_lots.append({"id": l["lot_id"], "segs": segs})

    days = ["9/22", "9/23", "9/24", "9/25", "9/26", "9/27", "9/28"]
    hm_vals = [[min(100, max(60, int(rnd.gauss(base, 6)))) for _ in days]
               for base in [96, 90, 84, 92, 78, 88, 94]]

    return {
        "status": {"labels": [ST[k] for k in ["run", "warn", "hold", "ok"]],
                   "keys": ["run", "warn", "hold", "ok"],
                   "values": [cnt[k] for k in ["run", "warn", "hold", "ok"]]},
        "wip": {"steps": STEPS[:-1], **wip},
        "timeline": {"start": "2026-09-23", "days": 8, "now": "2026-09-28T14:00", "lots": tl_lots},
        "heatmap": {"rows": STEPS, "cols": days, "values": hm_vals},
        "trend": {"weeks": ["W31", "W32", "W33", "W34", "W35", "W36", "W37", "W38", "W39"],
                  "rate": [78.2, 79.5, 81.0, 80.4, 82.9, 84.1, 83.6, 85.2, 86.4]},
    }


@app.route("/")
def lot_status():
    lots = build_lots()
    cnt = {k: sum(1 for l in lots if l["st"] == k) for k in ST}
    kpi = {"total": len(lots), "run": cnt["run"], "run_delta": 2, "warn": cnt["warn"],
           "hold": cnt["hold"], "hold_days": 2.5, "done_today": cnt["ok"], "done_week": 31,
           "auto_rate": 86.4, "auto_delta": 1.2}
    filters = {"products": PRODUCTS, "steps": STEPS}
    return render_template("lot_status.html", lots=lots, kpi=kpi, filters=filters,
                           page_data=build_page_data(lots), active_menu="lot",
                           updated_at="09/28 14:00", app_name="ER Monitor")


@app.route("/api/memo", methods=["POST"], endpoint="save_memo")
def save_memo():
    body = request.get_json(force=True)
    # TODO: 실제 저장 로직 (DB / 엑셀 / S3)
    print("memo saved:", body)
    return jsonify(ok=True)


if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8050, debug=True)
'''

  var v = function (name) { return css.getPropertyValue(name).trim(); };

  var T = {

    ink3:   v("--ink-3"),

  };

  var major = parseInt(String(Chart.version || "2").split(".")[0], 10);

  if (major >= 3) {

    d.font.size = 12;

    tt.cornerRadius = 4;
    tt.displayColors = true;

    d.elements.point.radius = 0;
    d.elements.point.hoverRadius = 4;

  function lerp(a, b, t) { return Math.round(a + (b - a) * t); }

  function scale(t) {

    var a = t < 0.5 ? STOPS[0] : STOPS[1], b = t < 0.5 ? STOPS[1] : STOPS[2], u = t < 0.5 ? t * 2 : (t - 0.5) * 2;
